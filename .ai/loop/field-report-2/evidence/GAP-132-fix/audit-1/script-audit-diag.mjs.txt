// GAP-132-fix audit-1: independent engine-level tab-close race probe.
// Real built engine (packages/browser/dist) in-process, real headless Chrome.
// Variants: closer = self (same puppeteer client) | external (a SECOND puppeteer.connect client
// closes the target, like a user/other agent closing the tab); timing = early (0-15ms, before
// the first poll), mid (20-420ms), final (close around the deadline of a 300ms wait),
// checkonce (timeoutMs:0, close 0-12ms); layout = single | iframe (same-origin iframe, #stay in
// main AND in iframe) | onlyiframe (#stay only in iframe).
// Instrumentation (prototype wrappers, not code changes): every pierceFirstMatch 'no-match' and
// every isHandleVisible 'not-visible' is a FALSE NEGATIVE here (#stay is always visible), so we
// count them per trial, plus every isTabClosed call's result.
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';
import http from 'node:http';
import { pathToFileURL } from 'node:url';
import { createRequire } from 'node:module';

const repoRoot = process.env.REPO;
const OUT = process.env.OUT;
const require_ = createRequire(path.join(repoRoot, 'packages', 'browser', 'package.json'));
const puppeteer = require_('puppeteer-core');
const eng = await import(pathToFileURL(path.join(repoRoot, 'packages', 'browser', 'dist', 'actions', 'browser-action-engine.js')));
const idx = await import(pathToFileURL(path.join(repoRoot, 'packages', 'browser', 'dist', 'index.js')));
const { BrowserActionEngine } = eng;
const delay = (ms) => new Promise((r) => setTimeout(r, ms));

const server = http.createServer((req, res) => {
  res.setHeader('content-type', 'text/html');
  const u = new URL(req.url, 'http://x');
  if (u.pathname === '/frame') return res.end('<div id="stay">visible in iframe</div>');
  if (u.pathname === '/iframe') return res.end('<div id="stay">always visible</div><iframe src="/frame"></iframe>');
  if (u.pathname === '/onlyiframe') return res.end('<div>main</div><iframe src="/frame"></iframe>');
  if (u.pathname === '/hiddenel') return res.end('<div id="stay" style="display:none">hidden</div>');
  return res.end('<div id="stay">always visible</div>');
});
await new Promise((r) => server.listen(0, '127.0.0.1', r));
const port = server.address().port;

let cur = null; // per-trial counters
const P = BrowserActionEngine.prototype;
const wrap = (name, onRet) => {
  const orig = P[name];
  P[name] = function (...args) {
    const r = orig.apply(this, args);
    if (r && typeof r.then === 'function') return r.then((v) => { if (cur) onRet(v, cur, args); return v; }, (e) => { if (cur) (cur.threw[name] = (cur.threw[name] ?? 0) + 1); throw e; });
    if (cur) onRet(r, cur, args);
    return r;
  };
};
wrap('pierceFirstMatch', (v, c) => { if (v?.kind === 'no-match' && c.closing) c.noMatchAfterClose++; if (v?.kind === 'no-match') c.noMatch++; });
wrap('isHandleVisible', (v, c) => { if (v === 'not-visible') c.notVisible++; });
wrap('isTabClosed', (v, c) => { c.tabClosedCalls.push(v); });

const profile = fs.mkdtempSync(path.join(os.tmpdir(), 'gap132-a1-diag-'));
const browser = await puppeteer.launch({ executablePath: process.env.CHROME_PATH ?? new idx.BrowserLauncher().findExecutablePath(), headless: true, userDataDir: profile, args: ['--no-sandbox'] });
const ext = await puppeteer.connect({ browserWSEndpoint: browser.wsEndpoint() });
const engine = new BrowserActionEngine();

const N = Number(process.env.N ?? 40);
const closers = (process.env.CLOSERS ?? 'self,external').split(',');
const timings = (process.env.TIMINGS ?? 'early,mid,final,checkonce').split(',');
const layouts = (process.env.LAYOUTS ?? 'single,iframe,onlyiframe').split(',');
const state = process.env.STATE ?? 'hidden';
const summary = [];
const falseSuccesses = [];
try {
  for (const layout of layouts) for (const closer of closers) for (const timing of timings) {
    let fs_ = 0, fail = 0, noMatchTotal = 0, noMatchAfterClose = 0, notVisTotal = 0, tabClosedTrue = 0, tabClosedFalse = 0;
    const msgs = {};
    for (let i = 0; i < N; i++) {
      const page = await browser.newPage();
      const pth = layout === 'single' ? '/' : '/' + layout;
      await page.goto(`http://127.0.0.1:${port}${pth}?i=${i}`);
      if (layout !== 'single') await page.waitForFunction(() => document.querySelector('iframe')?.contentDocument?.querySelector('#stay'), { timeout: 5000 }).catch(() => {});
      await delay(50);
      const targetId = page.target()._targetId;
      const tab = { id: `t${i}`, page, url: page.url(), title: '', recordAction() {} };
      let timeoutMs, closeDelay;
      if (timing === 'early') { timeoutMs = 2000; closeDelay = i % 16; }
      else if (timing === 'mid') { timeoutMs = 2000; closeDelay = 20 + ((i * 37) % 400); }
      else if (timing === 'final') { timeoutMs = 300; closeDelay = 240 + ((i * 7) % 140); }
      else { timeoutMs = 0; closeDelay = i % 12; }
      cur = { noMatch: 0, noMatchAfterClose: 0, notVisible: 0, tabClosedCalls: [], threw: {}, closing: false };
      const c = cur;
      const p = engine.executeAction(tab, { actionType: 'wait_for_selector', selector: '#stay', state, timeoutMs, maxRetries: 0 });
      if (closeDelay > 0) await delay(closeDelay);
      c.closing = true;
      if (closer === 'self') await page.close().catch(() => {});
      else {
        const t = (await ext.targets()).find((t) => t._targetId === targetId);
        const s = await ext.target().createCDPSession().catch(() => null);
        if (s) { await s.send('Target.closeTarget', { targetId }).catch(() => {}); await s.detach().catch(() => {}); }
        else if (t) { const pg = await t.page(); await pg?.close().catch(() => {}); }
      }
      const r = await p;
      cur = null;
      noMatchTotal += c.noMatch; noMatchAfterClose += c.noMatchAfterClose; notVisTotal += c.notVisible;
      tabClosedTrue += c.tabClosedCalls.filter(Boolean).length; tabClosedFalse += c.tabClosedCalls.filter((x) => !x).length;
      if (r.success) { fs_++; falseSuccesses.push({ layout, closer, timing, i, closeDelay, output: r.outputData, counters: c }); }
      else { fail++; const k = String(r.error).replace(/after \d+ms/, 'after Nms').slice(0, 200); msgs[k] = (msgs[k] ?? 0) + 1; }
      if (!page.isClosed()) await page.close().catch(() => {});
    }
    const row = { layout, closer, timing, state, trials: N, falseSuccess: fs_, failures: fail, pierceNoMatchTotal: noMatchTotal, pierceNoMatchAfterCloseStarted: noMatchAfterClose, handleNotVisibleTotal: notVisTotal, isTabClosedTrue: tabClosedTrue, isTabClosedFalse: tabClosedFalse, msgs };
    summary.push(row);
    console.log(JSON.stringify(row));
  }
} finally {
  fs.writeFileSync(OUT, JSON.stringify({ summary, falseSuccesses }, null, 2));
  await ext.disconnect().catch(() => {});
  await browser.close().catch(() => {});
  server.close();
  try { fs.rmSync(profile, { recursive: true, force: true }); } catch {}
}
process.exit(0);
